# 提示: DS 任务/工作流 state 字段常用值
#   0 SUBMITTED, 1 RUNNING, 2 READY_PAUSE, 3 PAUSE, 4 READY_STOP,
#   5 STOP, 6 FAILURE, 7 SUCCESS, 8 NEED_FAULT_TOLERANCE, 9 KILL

# 1. 任务表: 把非 SUCCESS/FAILURE 的任务标记为 FAILURE
UPDATE t_ds_task_instance
SET state = 6
WHERE state NOT IN (6, 7)
AND workflow_instance_name NOT LIKE '%cdc%';

# 2. 工作流表: 把非 SUCCESS/FAILURE 的工作流标记为 FAILURE
UPDATE t_ds_workflow_instance
SET state = 6
WHERE state NOT IN (6, 7)
AND name NOT IN ('XXX');
